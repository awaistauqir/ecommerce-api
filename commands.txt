#command to install redis in docker container

docker run -d -p 6379:6379 --name redis-cache redis:alpine

#command to create a migration file
npx migrate-mongo init

#command to create a migration file with name
npx migrate-mongo create add-default-sku-to-products

#command to run migration file
npx migrate-mongo up

#command to rollback migration file
npx migrate-mongo down


#stripe listen command
stripe login
copy the link provided by stripe login and paste it in your browser and login to your stripe account
paste the generated webhook secret key to your .env file in STRIPE_WEBHOOK_SECRET

#stripe command to listen to webhook events in development mode
stripe listen --forward-to localhost:3000/api/v1/webhooks/stripe

#command to run redis in docker container
docker start redis-cache

#command to stop redis in docker container
docker stop redis-cache

#command to remove redis in docker container
docker rm redis-cache

#command to run build and run the containeer
docker compose up --build

#shutdown the containeer
docker compose down

docker build -t ecommerce-api .
